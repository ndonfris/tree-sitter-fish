================================================================================
Line Continuation - command with comments
================================================================================

cmd \
   # first comment
   # second comment
   --flag \
   # third comment
   argument

--------------------------------------------------------------------------------

================================================================================
Line Continuation - Variable Assignments continued before a command
================================================================================

FIRST=one \

--------------------------------------------------------------------------------

================================================================================
Line Continuation - verify around pipes
================================================================================

first \
   # before the pipe
   | second | \
   # after the pipe
   # the backslash is continuation syntax, not a command name
   third

--------------------------------------------------------------------------------

(program
  (pipe
    (pipe
      (command
        name: (word)
        (comment))
      (command
        name: (word)))
    (comment)
    (comment)
    (command
      name: (word))))
